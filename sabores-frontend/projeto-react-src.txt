     1	import React from 'react';
     2	import { Navigate, useLocation } from 'react-router-dom';
     3	import { useAuth } from '../User/hooks/useAuth';
     4	import { ROUTES } from '../../Routes/constants';
     5	import LoadingSpinner from './LoadingSpinner';
     6	
     7	const PrivateRoute = ({ children, roles = [], redirectTo = ROUTES.LOGIN }) => {
     8	  const { isAuthenticated, loading, user } = useAuth();
     9	  const location = useLocation();
    10	
    11	  if (loading) {
    12	    return <LoadingSpinner />;
    13	  }
    14	
    15	  if (!isAuthenticated) {
    16	    // Redirecionar para login, mantendo a rota original para retorno
    17	    return <Navigate to={redirectTo} state={{ from: location }} replace />;
    18	  }
    19	
    20	  // Verificar roles se especificado
    21	  if (roles.length > 0) {
    22	    const hasRequiredRole = roles.some(role => user?.role === role);
    23	    
    24	    if (!hasRequiredRole) {
    25	      return <Navigate to={ROUTES.HOME} replace />;
    26	    }
    27	  }
    28	
    29	  return children;
    30	};
    31	
    32	export default PrivateRoute;

     1	import axios from 'axios';
     2	
     3	const API_URL = process.env.REACT_APP_API_URL || 'http://localhost:3000/api/v1';
     4	
     5	const authService = {
     6	  // Configuração do axios
     7	  axiosInstance: axios.create({
     8	    baseURL: API_URL,
     9	    headers: {
    10	      'Content-Type': 'application/json',
    11	    },
    12	  }),
    13	
    14	  // Interceptor para adicionar token
    15	  setupInterceptors: (store) => {
    16	    authService.axiosInstance.interceptors.request.use(
    17	      (config) => {
    18	        const token = localStorage.getItem('token');
    19	        if (token) {
    20	          config.headers.Authorization = `Bearer ${token}`;
    21	        }
    22	        return config;
    23	      },
    24	      (error) => Promise.reject(error)
    25	    );
    26	
    27	    // Interceptor para refresh token
    28	    authService.axiosInstance.interceptors.response.use(
    29	      (response) => response,
    30	      async (error) => {
    31	        const originalRequest = error.config;
    32	        
    33	        if (error.response?.status === 401 && !originalRequest._retry) {
    34	          originalRequest._retry = true;
    35	          
    36	          try {
    37	            const refreshToken = localStorage.getItem('refreshToken');
    38	            const response = await axios.post(`${API_URL}/users/refresh-token`, {
    39	              refreshToken,
    40	            });
    41	            
    42	            const { token } = response.data;
    43	            localStorage.setItem('token', token);
    44	            
    45	            originalRequest.headers.Authorization = `Bearer ${token}`;
    46	            return authService.axiosInstance(originalRequest);
    47	          } catch (refreshError) {
    48	            localStorage.removeItem('token');
    49	            localStorage.removeItem('refreshToken');
    50	            window.location.href = '/login';
    51	            return Promise.reject(refreshError);
    52	          }
    53	        }
    54	        
    55	        return Promise.reject(error);
    56	      }
    57	    );
    58	  },
    59	
    60	  // Métodos da API
    61	  login: async (email, password) => {

     2	const reportWebVitals = (metric) => {
     3	  // Use navigator.sendBeacon() or fetch() to send data to your analytics endpoint
     4	  const body = JSON.stringify(metric);
     5	  const url = 'https://your-analytics-endpoint.com';
     6	
     7	  // beacon is preferred as it doesn't delay page unloading
     8	  if (navigator.sendBeacon) {
     9	    navigator.sendBeacon(url, body);
    10	  } else {
    11	    fetch(url, {
    12	      body,
    13	      method: 'POST',
    14	      credentials: 'omit',
    15	      keepalive: true,
    16	    });
    17	  }
    18	};

     1	import React, { lazy, Suspense } from 'react';
     2	import { Routes, Route, Navigate } from 'react-router-dom';
     3	import { useAuth } from '../hooks/useAuth';
     4	import { ROUTES } from './constants';
     5	import AuthLayout from '../components/auth/AuthLayout';
     6	import LoadingSpinner from '../components/common/LoadingSpinner';
     7	
     8	// Lazy loading das páginas de autenticação
     9	const LoginPage = lazy(() => import('../pages/auth/Login'));
    10	const RegisterPage = lazy(() => import('../pages/auth/Register'));
    11	const RecoveryPage = lazy(() => import('../pages/auth/Recovery'));
    12	const ResetPasswordPage = lazy(() => import('../pages/auth/ResetPassword'));
    13	const VerifyEmailPage = lazy(() => import('../pages/auth/VerifyEmail'));
    14

     1	import React, { useState, useEffect } from 'react';
     2	import { Outlet, useLocation } from 'react-router-dom';
     3	import { Helmet } from 'react-helmet-async';
     4	import TopMenu from './Components/Top-Menu/TopMenu';
     5	import LeftMenu from './Components/Left-Menu/LeftMenu';
     6	import LeftMenuWrapper from './Components/Left-Menu/LeftMenuWrapper';

     9	import { useAuth } from './Components/User/hooks/useAuth';
    10	import './MainLayout.scss';
    11	
    12	const MainLayout = () => {
    13	  const location = useLocation();
    14	  const { loading } = useAuth();
    15	  const [isMobile, setIsMobile] = useState(false);
    16	  const [showLeftMenu, setShowLeftMenu] = useState(false);
    17	
    18	  useEffect(() => {
    19	    const checkMobile = () => {
    20	      setIsMobile(window.innerWidth <= 768);
    21	    };
    22	
    23	    checkMobile();
    24	    window.addEventListener('resize', checkMobile);
    25	
    26	    // Fechar menu left ao navegar (mobile)
    27	    setShowLeftMenu(false);
    28	
    29	    // Scroll para o topo
    30	    window.scrollTo(0, 0);
    31	
    32	    return () => {
    33	      window.removeEventListener('resize', checkMobile);
    34	    };
    35	  }, [location]);
    36	
    37	  const toggleLeftMenu = () => {
    38	    setShowLeftMenu(!showLeftMenu);
    39	  };
    40	
    41	  if (loading) {
    42	    return <LoadingSpinner fullScreen />;
    43	  }
    44	
    45	  return (
    46	    <>
    47	      <Helmet>
    48	        <html lang="pt-BR" />
    49	        <meta charSet="utf-8" />
    50	        <meta name="viewport" content="width=device-width, initial-scale=1.0, maximum-scale=5.0" />
    51	        <meta name="theme-color" content="#f97316" />
    52	        <link rel="manifest" href="/manifest.json" />
    53	        <link rel="icon" href="/favicon.ico" />
    54	        <link rel="apple-touch-icon" href="/logo192.png" />
    55	      </Helmet>
    56	
    57	      <div className="main-layout">
    58	        <TopMenu onMenuToggle={toggleLeftMenu} />
    59	        
    60	        <div className="main-layout__content">
    61	          {isMobile ? (
    62	            <>
    63	              {/* Overlay para menu mobile */}
    64	              {showLeftMenu && (
    65	                <div 
    66	                  className="main-layout__overlay"
    67	                  onClick={() => setShowLeftMenu(false)}
    68	                />
    69	              )}
    70	              
    71	              {/* Menu mobile */}
    72	              <div 
    73	                className={`main-layout__left-menu-mobile ${
    74	                  showLeftMenu ? 'main-layout__left-menu-mobile--open' : ''
    75	                }`}
    76	              >
    77	                <LeftMenu onClose={() => setShowLeftMenu(false)} />
    78	              </div>
    79	              
    80	              {/* Conteúdo principal mobile */}
    81	              <div className="main-layout__main">
    82	                <Outlet />
    83	              </div>
    84	            </>
    85	          ) : (
    86	            // Desktop layout com wrapper
    87	            <LeftMenuWrapper>
    88	              <div className="main-layout__main">
    89	                <Outlet />
    90	              </div>
    91	            </LeftMenuWrapper>
    92	          )}
    93	        </div>
    94	
    95	        {/*<Footer />*/}
    96	      </div>
